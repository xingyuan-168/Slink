cmake_minimum_required(VERSION 3.25)
project(CQ_SLink VERSION 1.0.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "CQ_SLink requires Windows")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /permissive- /utf-8 /EHsc)
enable_testing()
add_executable(test_source WIN32 tests/test_source.cpp)
target_link_libraries(test_source PRIVATE user32 gdi32 d3d11 dxgi d2d1 dwrite)
add_executable(dwm_poc tests/dwm_poc.cpp)
target_link_libraries(dwm_poc PRIVATE user32 gdi32 dwmapi d3d11 dxgi windowsapp)
add_test(NAME poc_gdi COMMAND dwm_poc --backend gdi --cycles 2 --seconds 2)
add_test(NAME poc_d3d COMMAND dwm_poc --backend d3d --cycles 2 --seconds 2)
add_test(NAME poc_wgc_gdi COMMAND dwm_poc --wgc --backend gdi --cycles 2 --seconds 2)
add_test(NAME poc_wgc_d3d COMMAND dwm_poc --wgc --backend d3d --cycles 2 --seconds 2)
set_tests_properties(poc_gdi poc_d3d poc_wgc_gdi poc_wgc_d3d PROPERTIES RUN_SERIAL TRUE TIMEOUT 60)
