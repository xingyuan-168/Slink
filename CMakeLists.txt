cmake_minimum_required(VERSION 3.25)
project(GameWindowMapper VERSION 1.0.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "GameWindowMapper requires Windows and MSVC")
endif()
option(GWM_BUILD_DLL "Build the x86 DLL, demo and API tests" ON)
option(GWM_BUILD_TESTS "Build native API tests and dynamic capture sources" ON)
option(GWM_BUILD_HELPER "Build the native x64 virtual display recovery helper" OFF)
option(GWM_ENABLE_DISPLAY_TESTS "Opt in to tests that temporarily enable only the installed own virtual display" OFF)
set(GWM_VENDOR_DIR "${CMAKE_CURRENT_SOURCE_DIR}/out/vendor/virtual-display-driver/25.7.23" CACHE PATH "Pinned original VirtualDrivers driver files")
set(GWM_HELPER_EXE "${CMAKE_CURRENT_SOURCE_DIR}/out/build/x64-helper/Release/gwm_recovery.exe" CACHE FILEPATH "Native x64 helper embedded into the DLL")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/W4;/permissive-;/utf-8;/EHsc>")
set(GWM_WINDOWS_LIBS d3d11 dxgi d2d1 dwrite dwmapi windowscodecs runtimeobject windowsapp user32 gdi32 ole32 shcore)
if(GWM_BUILD_HELPER)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_GENERATOR_PLATFORM STREQUAL "x64")
    message(FATAL_ERROR "gwm_recovery requires a native AMD64 build: use the x64-helper preset.")
  endif()
  add_executable(gwm_recovery
    src/virtual/helper_main.cpp src/virtual/broker.cpp src/virtual/protocol.cpp src/virtual/recovery_prompt.cpp
    src/virtual/window_parking.cpp src/virtual/display_manager.cpp resources/test_host.rc)
  target_include_directories(gwm_recovery PRIVATE src resources)
  target_link_options(gwm_recovery PRIVATE /MANIFEST:NO)
  target_link_libraries(gwm_recovery PRIVATE user32 gdi32 advapi32 ole32 shcore shell32 setupapi newdev cfgmgr32 bcrypt wintrust uuid)
  add_executable(driver_probe tests/driver_probe.cpp src/virtual/display_manager.cpp)
  target_include_directories(driver_probe PRIVATE src)
  target_link_libraries(driver_probe PRIVATE setupapi newdev cfgmgr32 shell32 ole32 advapi32 bcrypt wintrust user32 uuid)
  enable_testing()
  add_executable(recovery_prompt_tests tests/recovery_prompt_tests.cpp src/virtual/recovery_prompt.cpp resources/test_host.rc)
  target_include_directories(recovery_prompt_tests PRIVATE src resources)
  target_link_options(recovery_prompt_tests PRIVATE /MANIFEST:NO)
  target_link_libraries(recovery_prompt_tests PRIVATE user32 gdi32)
  add_test(NAME recovery_prompt COMMAND recovery_prompt_tests)
  add_executable(request_cache_tests tests/request_cache_tests.cpp src/virtual/broker.cpp src/virtual/protocol.cpp
    src/virtual/window_parking.cpp src/virtual/display_manager.cpp src/virtual/recovery_prompt.cpp)
  target_include_directories(request_cache_tests PRIVATE src)
  target_link_libraries(request_cache_tests PRIVATE user32 gdi32 advapi32 ole32 shcore shell32 setupapi newdev cfgmgr32 bcrypt wintrust uuid)
  add_test(NAME request_cache COMMAND request_cache_tests)
endif()
if(GWM_BUILD_DLL)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "The public HWND ABI requires x86: use the x86 preset. For x64 source windows use -DGWM_BUILD_DLL=OFF.")
  endif()
  if(NOT EXISTS "${GWM_HELPER_EXE}")
    message(FATAL_ERROR "Build the x64-helper Release target before configuring the DLL. scripts/build.ps1 performs both stages.")
  endif()
  set(GWM_VENDOR_FILES MttVDD.inf MttVDD.dll mttvdd.cat)
  set(GWM_VENDOR_HASHES
    550d211fe481e74dfe3f9d724ed78be48b3a9113405965d683d9373e8d672f5d
    c9ca837f57a98fbd43bc416a7f535a95843626e7759eaf85cf0cd7ce334dbb05
    08a0093fc9b2e32b287a6f8a77ca4de0a31830d29fc33d2b13a918dc859468f6)
  foreach(GWM_FILE GWM_EXPECTED IN ZIP_LISTS GWM_VENDOR_FILES GWM_VENDOR_HASHES)
    if(NOT EXISTS "${GWM_VENDOR_DIR}/${GWM_FILE}")
      message(FATAL_ERROR "Missing original ${GWM_FILE}: run scripts/fetch-driver.ps1 first.")
    endif()
    file(SHA256 "${GWM_VENDOR_DIR}/${GWM_FILE}" GWM_ACTUAL)
    if(NOT GWM_ACTUAL STREQUAL GWM_EXPECTED)
      message(FATAL_ERROR "Pinned SHA256 mismatch for ${GWM_FILE}; refusing to embed modified driver bytes.")
    endif()
  endforeach()
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
  file(SHA256 "${GWM_HELPER_EXE}" GWM_HELPER_HASH)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/third_party/VirtualDisplayDriver/LICENSE.txt" GWM_LICENSE_HASH)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/payload_hashes.h"
    "#pragma once\n#define GWM_PAYLOAD_HELPER_SHA256 \"${GWM_HELPER_HASH}\"\n#define GWM_PAYLOAD_LICENSE_SHA256 \"${GWM_LICENSE_HASH}\"\n")
  configure_file(resources/payload.rc.in "${CMAKE_CURRENT_BINARY_DIR}/generated/payload.rc" @ONLY)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${GWM_HELPER_EXE}" "${GWM_VENDOR_DIR}/MttVDD.inf" "${GWM_VENDOR_DIR}/MttVDD.dll" "${GWM_VENDOR_DIR}/mttvdd.cat"
    "${CMAKE_CURRENT_SOURCE_DIR}/third_party/VirtualDisplayDriver/LICENSE.txt")
  set_property(SOURCE "${CMAKE_CURRENT_BINARY_DIR}/generated/payload.rc" PROPERTY OBJECT_DEPENDS
    "${GWM_HELPER_EXE};${GWM_VENDOR_DIR}/MttVDD.inf;${GWM_VENDOR_DIR}/MttVDD.dll;${GWM_VENDOR_DIR}/mttvdd.cat;${CMAKE_CURRENT_SOURCE_DIR}/third_party/VirtualDisplayDriver/LICENSE.txt")
  add_library(GameWindowMapper SHARED src/exports.cpp src/engine.cpp src/capture.cpp src/graphics.cpp
    src/virtual/payload.cpp src/virtual/client.cpp src/virtual/protocol.cpp src/virtual/controller.cpp
    resources/mapper.rc resources/exports.def "${CMAKE_CURRENT_BINARY_DIR}/generated/payload.rc")
  target_compile_definitions(GameWindowMapper PRIVATE GWM_BUILD)
  target_include_directories(GameWindowMapper PUBLIC include PRIVATE src resources "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_libraries(GameWindowMapper PRIVATE ${GWM_WINDOWS_LIBS} bcrypt advapi32 shell32 uuid)
  add_executable(mapper_demo examples/mapper_demo.cpp resources/test_host.rc)
  target_link_options(mapper_demo PRIVATE /MANIFEST:NO)
  target_include_directories(mapper_demo PRIVATE resources)
  target_link_libraries(mapper_demo PRIVATE GameWindowMapper user32)
  # The delivery is exactly one DLL and one self-contained usage guide.
  install(FILES "$<TARGET_FILE:GameWindowMapper>" DESTINATION .)
  install(FILES docs/E_LANGUAGE.md DESTINATION . RENAME 使用说明.md)
endif()
