cmake_minimum_required(VERSION 3.25)
project(GameWindowMapper VERSION 1.0.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "GameWindowMapper requires Windows and MSVC")
endif()
option(GWM_BUILD_DLL "Build the x86 DLL, demo and API tests" ON)
option(GWM_BUILD_TESTS "Build native API tests and dynamic capture sources" ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/W4;/permissive-;/utf-8;/EHsc>")
set(GWM_WINDOWS_LIBS d3d11 dxgi d2d1 dwrite dwmapi windowscodecs runtimeobject windowsapp user32 gdi32 ole32 shcore)
if(GWM_BUILD_DLL)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "The public HWND ABI requires x86: use the x86 preset. For x64 source windows use -DGWM_BUILD_DLL=OFF.")
  endif()
  add_library(GameWindowMapper SHARED src/exports.cpp src/engine.cpp src/capture.cpp src/graphics.cpp resources/mapper.rc resources/exports.def)
  target_compile_definitions(GameWindowMapper PRIVATE GWM_BUILD)
  target_include_directories(GameWindowMapper PUBLIC include PRIVATE src resources)
  target_link_libraries(GameWindowMapper PRIVATE ${GWM_WINDOWS_LIBS})
  add_executable(mapper_demo examples/mapper_demo.cpp resources/test_host.rc)
  target_link_options(mapper_demo PRIVATE /MANIFEST:NO)
  target_include_directories(mapper_demo PRIVATE resources)
  target_link_libraries(mapper_demo PRIVATE GameWindowMapper user32)
  install(TARGETS GameWindowMapper mapper_demo RUNTIME DESTINATION . ARCHIVE DESTINATION lib)
  install(FILES include/GameWindowMapper.h DESTINATION include)
  install(FILES resources/GameWindowMapper.ico DESTINATION .)
  install(FILES docs/E_LANGUAGE.md DESTINATION docs)
  install(DIRECTORY examples/ DESTINATION examples)
endif()
if(GWM_BUILD_TESTS)
  enable_testing()
  add_executable(dynamic_source tests/dynamic_source.cpp resources/test_host.rc)
  target_link_options(dynamic_source PRIVATE /MANIFEST:NO)
  target_include_directories(dynamic_source PRIVATE resources)
  target_link_libraries(dynamic_source PRIVATE ${GWM_WINDOWS_LIBS} d3dcompiler)
  if(GWM_BUILD_DLL)
    add_library(GameWindowMapper_test SHARED src/exports.cpp src/engine.cpp src/capture.cpp src/graphics.cpp resources/mapper.rc resources/exports.def)
    target_compile_definitions(GameWindowMapper_test PRIVATE GWM_BUILD GWM_TEST_HOOKS)
    target_include_directories(GameWindowMapper_test PRIVATE include src resources)
    target_link_libraries(GameWindowMapper_test PRIVATE ${GWM_WINDOWS_LIBS})
    add_executable(native_tests tests/native_tests.cpp resources/test_host.rc)
    target_link_options(native_tests PRIVATE /MANIFEST:NO)
    target_include_directories(native_tests PRIVATE include resources)
    target_link_libraries(native_tests PRIVATE user32 gdi32 psapi)
    add_dependencies(native_tests GameWindowMapper GameWindowMapper_test dynamic_source)
    add_test(NAME native_api COMMAND native_tests --smoke)
    add_executable(gpu_geometry_tests tests/gpu_geometry_tests.cpp src/graphics.cpp)
    target_include_directories(gpu_geometry_tests PRIVATE src include)
    target_link_libraries(gpu_geometry_tests PRIVATE ${GWM_WINDOWS_LIBS})
    add_test(NAME gpu_geometry COMMAND gpu_geometry_tests)
    add_test(NAME device_recovery COMMAND native_tests --smoke --require-device-hook --dll $<TARGET_FILE:GameWindowMapper_test>)
    set_tests_properties(device_recovery PROPERTIES RUN_SERIAL TRUE TIMEOUT 180)
    set_tests_properties(native_api PROPERTIES RUN_SERIAL TRUE TIMEOUT 180)
    install(TARGETS native_tests dynamic_source RUNTIME DESTINATION .)
  endif()
endif()
